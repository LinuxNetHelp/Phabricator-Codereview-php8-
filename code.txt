CREATE DATABASE phabricator;
CREATE USER 'phab_user'@'localhost' IDENTIFIED BY 'password@123';
GRANT ALL PRIVILEGES ON `phabricator\_%`.* TO 'phab_user'@'localhost';
GRANT ALL PRIVILEGES ON *.* TO 'phab_user'@'localhost';

exit

cd /opt/phabricator

sudo ./bin/storage upgrade --user user --password yourpassword
sudo ./bin/config set phabricator.base-uri 'http://domain or ip /'
